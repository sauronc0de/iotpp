cmake_minimum_required(VERSION 3.22)
set(APP_NAME "sentinel" CACHE STRING "Application to build")
set(FIRMWARE_RELEASE "0.0.0" CACHE STRING "Release version (major.minor.patch; components 0..255)")
if(NOT FIRMWARE_RELEASE MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "FIRMWARE_RELEASE must be major.minor.patch")
endif()
string(REPLACE "." ";" release_components "${FIRMWARE_RELEASE}")
foreach(component IN LISTS release_components)
    if(component GREATER 255 OR component MATCHES "^0[0-9]+")
        message(FATAL_ERROR "Release components must be canonical integers in 0..255")
    endif()
endforeach()
include("${CMAKE_CURRENT_SOURCE_DIR}/tasks/discover.cmake")
set(SDKCONFIG "${CMAKE_CURRENT_SOURCE_DIR}/config/build/sdkconfig" CACHE FILEPATH "Local project configuration")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_SOURCE_DIR}/config/sdkconfig.defaults" CACHE STRING "Shared project defaults")
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/app/${APP_NAME}")

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(IOTPP_AVAILABLE_TASKS "${available_tasks}")
idf_build_set_property(IOTPP_TASK_KCONFIG "${task_kconfig_path}")

set(PROJECT_VER "${FIRMWARE_RELEASE}")
project(${APP_NAME} VERSION ${FIRMWARE_RELEASE})

foreach(library dex dex_server rbamp_reading monitoring)
    add_subdirectory("../libs/${library}" "${CMAKE_CURRENT_BINARY_DIR}/libs/${library}" EXCLUDE_FROM_ALL)
endforeach()

string(REPLACE "." "," firmware_version "${PROJECT_VERSION}")
idf_component_get_property(app_component ${APP_NAME} COMPONENT_LIB)
target_compile_definitions(${app_component} PRIVATE "FIRMWARE_VERSION=${firmware_version}")
